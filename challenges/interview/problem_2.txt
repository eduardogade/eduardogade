# ----- Constants -----

# I "cheated" and look this up because I haven't memorized
_BOMS: Final[tuple[str, ...]] = (
    "\ufeff",
)

# I "cheated" and look this up because I haven't memorized
_LINE_SEPARATORS: Final[tuple[str, ...]] = (
    "\r\n", "\n", "\r",
    "\u2028", "\u2029",
    " ", "\t",
)

_ALL_DIACRYTICS: Final[tuple[str, ...]] = (
    "\ufeff",
    "\r\n", "\n", "\r",
    "\u2028", "\u2029",
    " ", "\t",
)


# ----- Functions -----

def clean_word(word: str) -> str:
    """
    - Removes BOMs.
    - Removes newline-like symbols.
    """
    new_word: str = ""

    # Remove BOM only at the beginning
    new_word = "".join([char for char in word if char not in _ALL_DIACRYTICS])

    return new_word


def word_groupping(input_list: list[str]) -> dict[str, list[str]]:
    """
    1. Create output and helper elements.
    2. Iterates on input.
    3. Clean input word.
    4. Verify identicals with hashmap.
    5. Append cleaned word to dictionary if new.
    """

    grouped_dict: dict[str, list[str]] = dict()

	# Helper hashmap
    helper_hashmap: dict[str, bool] = dict()

    for word in input_list:

        # Clean
        cleaned_word = clean_word(word.lower())
        if not cleaned_word:
            continue # Needed since it may return an empty string and further ops require a string

        # Verify if word is in hashmap
        if helper_hashmap.get(cleaned_word, None):
            continue
        helper_hashmap[cleaned_word] = True

        initial = cleaned_word[0]
        grouped_dict[initial] = grouped_dict.get(initial, []).append(cleaned_word)

    # Returning the output object
    return grouped_dict
